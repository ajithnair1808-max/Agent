public static async dispatchToWebhook(
  webhookUrl: string,
  payload: {
    type: string;
    title: string;
    estimatedValue: number;
    content: string;
    timestamp: string;
  }
): Promise<{ success: boolean; message: string }> {
  if (!webhookUrl || !webhookUrl.trim()) {
    return { success: false, message: 'Webhook URL not configured.' };
  }

  try {
    if (webhookUrl.includes('discord.com/api/webhooks')) {
      await fetch(webhookUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          username: 'Agentic Wealth Bot',
          embeds: [ { title: `⚡ [AUTOMATED DISPATCH] ${payload.title}`, description: payload.content.substring(0, 1000) + '...', ... } ]
        }),
      });
    } else {
      await fetch(webhookUrl, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });
    }

    return { success: true, message: 'Dispatched successfully via automated webhook!' };
  } catch (error: any) {
    return { success: false, message: `Webhook notice: Payload prepared. (${error.message || 'CORS sandbox'})` };
  }
}
